\begin{frame}{Data Persistence}
    \textbf{Persistence} the period of time during which an object is usable\footnote{M.~P. Atkinson, P.~J. Bailey, K.~J. Chisholm, W.~P. Cockshott, and R.~Morrison, ``Ps-algol: A language for persistent programming,'' 1983.}

    \bigskip
    Today's dominant model --- \textbf{two-level store} (non-persistent):
    \begin{itemize}
        \item Data is either \textit{active} (RAM) or \textit{stored} (disk)
        \item Programmer manually moves data between levels
        
    \end{itemize}
    \bigskip

    Consequences:
    \begin{itemize}
        \item Data loss
        \item I/O boilerplate ($\approx$30\% of code)
        \item Frequent initialization and configuration
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Orthogonal Persistence}
    Alternative: the \textbf{environment} manages persistence

    \bigskip
    Principles of \textbf{orthogonal persistence}\footnote{R.~Morrison and M.~P. Atkinson, ``Persistent Languages and Architectures,'' 1990.}:
    \begin{enumerate}
        \item \textbf{Independence}
        \item \textbf{Data type orthogonality}
        \item \textbf{Identification}
    \end{enumerate}

    \bigskip
    \textbf{Result:} No two-level store model's drawbacks and improved disk's bandwidth utilization
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{PhantomOS}
    \textbf{PhantomOS}\footnote{\url{http://phantomos.org/}} is a non-Unix like OS implementing orthogonal persistence.\footnote{D.~Zavalishin, PhantomOS, 2010.}

    \bigskip
    \begin{itemize}
        \item Phantom Virtual Machine (PVM): bytecode VM of a persistent language
        \item Simplified POSIX layer (non-persistent)
        \item Periodical snapshots (via \textbf{Copy-on-Write})
        \item Changed pages written to disk (incremental)
        \item Programs are temporarily suspended (snapshot preparation)
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Genode}
    PhantomOS ported to the \textbf{Genode} OS C++ framework\footnote{\url{https://genode.org/}}:
    \begin{itemize}
        \item Proven microkernels (NOVA, seL4, Fiasco.OC)
        \item Capability-based security
        \item Minimal TCB
    \end{itemize}

    \bigskip
\end{frame}

\begin{frame}{Snapper}
    \textbf{Snappser}\footnote{\url{https://github.com/rumenmitov/snapper}} is a file-system-based transactional snapshot mechanism\footnote{R.~Mitov, ``Snapper (v2): A PhantomOS snapshot mechanism,'' FOSDEM 2026. \url{https://fosdem.org/2026/schedule/event/DJ8YVT-practical-persistence/}}

    \bigskip
    Previous approach (``superblock''): snapshots stored as monolithic objects.
    \begin{itemize}
        \item[$+$] Fast
        \item[$-$] No fault tolerance, no storage recovery
    \end{itemize}

    \bigskip
    Snapper 2.0:
    \begin{itemize}
        \item Configured generations number
        \item Unchanged pages \textbf{linked}, not copied across generations
        \item ext4 + fsck for crash recovery
        \item Integrity checks (hashes per file)
        \item Configurable redundancy policy
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Problem Statement}
    Problem:
    \begin{itemize}
        \item Snapshots contain \textbf{sensitive data}
        \item \textbf{Plaintext} snapshots are stored on the drive
    \end{itemize}
    \bigskip
    
    Consequences:
    \begin{itemize}
        \item Data can be \textbf{read directly}
        \item Ability to \textbf{restore the complete state}
    \end{itemize}
    \bigskip

    Immediate decision's challenges:
    \begin{itemize}
        \item \textbf{No prior research} on this for PhantomOS
        \item \textbf{No ready-made solution} exists in Genode
    \end{itemize}

    \bigskip
    \textbf{Solution}: to implement a mechanism for the secure storage of snapshots.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Environment \& Requirements}
    No unauthorized access to data

    \bigskip
    Requirements:
    \begin{enumerate}
        \item Mandatory authorization
        \item Key stored on an \textbf{external USB drive}
        \item Encryption of snapshots (AES, block-level)
        \item Only encrypted data written to disk
        \item Acceptable performance overhead ($\leq 2\times$)
    \end{enumerate}

    \bigskip
    Requirements 1 \& 2 combined: the USB drive contains the key
    and serves as the authentication factor.
\end{frame}

\begin{frame}{Implementation: se\_vault}
    \textbf{file\_vault} based on \textbf{Tresor} library:
    \begin{itemize}
        \item Tresor: AES encryption on 4K blocks, SHA-256 integrity, CoW atomicity
        \item file\_vault: state orchestrator providing a secure FS session
    \end{itemize}

    \bigskip
    Implemented \textbf{se\_vault}\footnote{\url{https://github.com/VoronM1522/genode/tree/se/repos/gems/src/app/se_vault}}
    based on \textbf{file\_vault}\footnote{\url{https://github.com/genodelabs/genode/tree/sculpt-25.04/repos/gems/src/app/file_vault}}:
    \begin{itemize}
        \item Headless operation
        \item Passphrase replaced by \textbf{USB key file}
        \item rump\_fs/ext2 replaced by \textbf{lwext4/ext4}
        \item Added \textbf{fsck} crash recovery support with fsck
        \item Direct block session (skip image file creation)
        \item All block sizes standardized to 4K
        \item Graceful shutdown sequence added
    \end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Related Fixes \& Adaptations}
    \textbf{Functionality patches:}
    \begin{itemize}
        \item lwext4: \texttt{ftruncate} and \texttt{sync}
        \item isomem: power-off button callback and graceful shutdown
        \item snapper: client counter and graceful shutdown
        \item vfs: client counter, graceful shutdown and clean unmount
    \end{itemize}

    \bigskip
    \textbf{Optimization}: using a "Block" session directly\\
    Reduced initialization time from $\approx$20 min to $\approx$1 min
\end{frame}
